@Mapper注解是需要在启动类中用@MapperScan("")注解去精确扫描的


查询涉及到多个表时, 要不就采用多表联查left join然后再\<resultMap中用\<association去根据字段注入
要不就不用多表查, 选择在\<association中去使用别的Mapper接口中的方法去获得结果然后注入, 我指的就是这部分
```
<!--    我直接在这里封装一个多sql查询直接给VoComment中的VoUser注入了-->  
    <select id="selectCommentByArticleId" resultMap="VoCommentMP">  
        select <include refid="CommentAll"></include> from tb_comment where article_id = #{id}  
    </select>  
    <!--    需要一个中间的查询user信息的sql-->  
    <resultMap id="VoCommentMP" type="com.Project.blog.Bean.Vo.Comment.VoComment">  
        <id column="id" property="id"></id>  
        <result column="context" property="context"></result>  
        <association property="voUser"  
                     resultMap="com.Project.blog.mapper.UserMapper.VoUserMP"  
                     select="com.Project.blog.mapper.UserMapper.selectById"  
                     column="user_id"></association>  
    </resultMap>
```
