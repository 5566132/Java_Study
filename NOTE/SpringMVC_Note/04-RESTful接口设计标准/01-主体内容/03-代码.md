`package com.atguigu.rest.crud.Controller;`  
  
`import com.atguigu.rest.crud.R;`  
`import com.atguigu.rest.crud.Service.EmplayeeService;`  
`import com.atguigu.rest.crud.bean.Employee;`  
`import org.springframework.beans.factory.annotation.Autowired;`  
`import org.springframework.web.bind.annotation.*;`  
  
`import java.util.List;`  
  
`@CrossOrigin//允许跨域, 不同源无所谓`  
`@RequestMapping("/api/v1")`  
`@RestController`  
`public class EmployeeRestController {`  
    `@Autowired`  
    `private EmplayeeService es;`  
  
    `//id查单个员工`  
    `@GetMapping (value = "/employee/{id}")`  
    `public R<Employee> get(@PathVariable("id") Long id) {`  
        `return R.ok(es.getEmployee(id));`  
    `}`  
  
    `//查所有员工`  
    `@GetMapping("/employees")`  
    `public R<List<Employee>> getAll(){`  
        `return R.ok(es.getAllEmployees());`  
    `}`  
  
  
    `//增加员工`  
    `@PostMapping("/employee")`  
    `public R add(@RequestBody Employee emp){`  
        `es.addEmployee(emp);`  
        `return R.ok();`  
    `}`  
  
    `//删除`  
    `@DeleteMapping("/employee/{id}")`  
    `public R delete(@PathVariable Long id){`  
        `es.deleteEmployee(id);`  
        `return R.ok();`  
    `}`  
  
    `//更改`  
    `@PutMapping("/employee")`  
    `public R update(@RequestBody Employee emp){`  
        `es.updateEmployee(emp);`  
        `return R.ok();`  
    `}`  
  
  
  
  
`}`
